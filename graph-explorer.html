# Finde alle Wohnungen mit GEAK-Klasse ≥ D und offenem Mietverhältnis
SELECT ?apt ?storey ?tenant ?rent WHERE {
  ?apt   a rec:Apartment ;
         rec:locatedIn ?storey ;
         rec:energyClass ?class .
  ?lease a rec:Lease ;
         rec:forSpace ?apt ;
         rec:hasTenant ?tenant ;
         rec:monthlyRent ?rent ;
         rec:status "active" .
  FILTER(?class IN ("D","E","F","G"))
}
ORDER BY DESC(?rent)
